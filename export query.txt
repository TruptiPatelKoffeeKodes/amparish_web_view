SELECT sku,PurchaseAmt
FROM (
    SELECT *, ROW_NUMBER() OVER (PARTITION BY sku ORDER BY PurchaseDate DESC) AS rn
    FROM purchase
    WHERE sku IN ("AENMB0DDV3V6JV", "AENMB07Y51PCW7")
) t
WHERE t.rn = 1;

//other query of this
SELECT p.sku,p.PurchaseAmt
FROM purchase p
JOIN (
    SELECT sku, MAX(PurchaseDate) AS max_date
    FROM purchase
    WHERE sku IN ("AENMB0DDV3V6JV", "AENMB07Y51PCW7")
    GROUP BY sku
) latest ON p.sku = latest.sku AND p.PurchaseDate = latest.max_date;
